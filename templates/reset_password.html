import socket, requests, time, os, win32gui, win32con, win32api from PIL import ImageGrab from pynput import mouse, keyboard from datetime import datetime SERVER_URL = "https://rmm-cloud-server.onrender.com" HOSTNAME = socket.gethostname() SHARED_NETWORK_PATH = r"\\172.18.128.65\Screenshort" last_activity_time = time.time() last_sent_title = "" blocked_policies = [] def on_activity(*args): global last_activity_time last_activity_time = time.time() mouse_listener = mouse.Listener(on_move=on_activity, on_click=on_activity, on_scroll=on_activity) key_listener = keyboard.Listener(on_press=on_activity) mouse_listener.start() key_listener.start() def get_active_window_info(): try: hwnd = win32gui.GetForegroundWindow() title = win32gui.GetWindowText(hwnd) return hwnd, title except Exception: return None, "" def enforce_policy(): global blocked_policies hwnd, title = get_active_window_info() if not title: return False title_lower = title.strip().lower() for b in blocked_policies: if not b: continue clean_keyword = b.strip().lower().replace("https://", "").replace("http://", "").replace("www.", "").replace(".com", "").replace(".in", "").replace(".org", "") if clean_keyword and clean_keyword in title_lower: print(f"[POLICY VIOLATION] Closing tab for window: '{title}' matching keyword '{clean_keyword}'") try: win32api.keybd_event(0x11, 0, 0, 0) # Ctrl DOWN win32api.keybd_event(0x57, 0, 0, 0) # W DOWN time.sleep(0.05) win32api.keybd_event(0x57, 0, 2, 0) # W UP win32api.keybd_event(0x11, 0, 2, 0) # Ctrl UP time.sleep(0.5) except Exception as e: print(f"[Close Tab Error] {e}") return True return False def sync_history(): global last_sent_title hwnd, title = get_active_window_info() title_clean = title.strip() if title_clean and title_clean != last_sent_title: if not title_clean.startswith("C:/") and "Command Prompt" not in title_clean: try: requests.post(f"{SERVER_URL}/api/system/history", json={"hostname": HOSTNAME, "url": title_clean}, timeout=2) last_sent_title = title_clean except Exception: pass def run(): global blocked_policies print(f"RMM Agent Running for PC: {HOSTNAME}...") pc_folder = os.path.join(SHARED_NETWORK_PATH, HOSTNAME) try: os.makedirs(pc_folder, exist_ok=True) except Exception: pass last_ping_time = 0 while True: try: now = time.time() idle_seconds = now - last_activity_time # Status determination screenshot_status = "Sending Screenshots" if idle_seconds <= 10 else "Paused (User Idle)" if now - last_ping_time >= 2: res = requests.post( f"{SERVER_URL}/api/system/ping", json={"hostname": HOSTNAME, "screenshot_status": screenshot_status}, timeout=2 ) if res.ok: blocked_policies = res.json().get("policies", []) last_ping_time = now was_closed = enforce_policy() if not was_closed: sync_history() # Active capture logic if idle_seconds <= 10: shot = ImageGrab.grab() now_str = datetime.now().strftime("%Y%m%d_%H%M%S") filename = f"{HOSTNAME}_{now_str}.png" filepath = os.path.join(pc_folder, filename) shot.save(filepath) except Exception as e: print(f"[Error] {e}") time.sleep(0.5) if __name__ == "__main__": run()
