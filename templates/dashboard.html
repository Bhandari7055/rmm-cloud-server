document.getElementById('blockForm').addEventListener('submit', function(e) { e.preventDefault(); const domainInput = document.getElementById('domainInput'); const domain = domainInput.value.trim(); if(!domain) return; fetch('/api/blocked-domains', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ domain: domain, action: 'block' }) }) .then(res => res.json()) .then(data => { if(data.status === 'success') { domainInput.value = ''; // Directly add domain to UI list without hard page reload const ul = document.getElementById('blockedDomainsList'); if(ul.children[0] && ul.children[0].innerText.includes('No domains currently blocked')) { ul.innerHTML = ''; } const li = document.createElement('li'); li.className = 'list-group-item d-flex justify-content-between align-items-center'; li.innerHTML = `${domain} `; ul.appendChild(li); } }); }); function unblockDomain(domain) { fetch('/api/blocked-domains', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ domain: domain, action: 'unblock' }) }) .then(res => res.json()) .then(data => { if(data.status === 'success') { location.reload(); } }); }
