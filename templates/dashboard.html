function viewHistory(hostname) { document.getElementById('historyModalTitle').innerText = "Browsing History: " + hostname; historyModalObj = new bootstrap.Modal(document.getElementById('historyModal')); historyModalObj.show(); fetch('/api/system/history?hostname=' + encodeURIComponent(hostname)) .then(res => res.json()) .then(data => { const tbody = document.getElementById('historyTableBody'); tbody.innerHTML = ''; if(!data.history || data.history.length === 0) { tbody.innerHTML = 'No history logged yet.'; return; } data.history.slice().reverse().forEach(item => { const tr = document.createElement('tr'); let urlDisplay = item.url; let targetLink = item.url; // Fix 404 issue by routing non-HTTP text to Google Search if (!item.url.startsWith('http://') && !item.url.startsWith('https://')) { targetLink = 'https://www.google.com/search?q=' + encodeURIComponent(item.url); } tr.innerHTML = ` ${urlDisplay} ${item.timestamp} `; tbody.appendChild(tr); }); }); }
